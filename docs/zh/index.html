    npx skills add NB-Corp/nb-speak --global --agent '*'
    
          
    

          
    
            
    
              2. OpenClaw（Telegram · WhatsApp · Discord）
              
            
    
            
    npx --yes github:NB-Corp/nb-speak openclaw --apply
